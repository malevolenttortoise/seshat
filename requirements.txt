# Cover image perceptual hashing for MAM URL verification (Part C).
# imagehash provides pHash (DCT-based); Pillow is its image backend.
# Pinning Pillow here makes the runtime contract explicit. ~3MB on the slim
# image. Validated 2026-05-09 against 16 image pairs (right-Possibles
# cluster at distance 0-6, wrong-matches at 28-36).
# 13.0.0 clears 28 advisories open against 12.3.0 (56 Dependabot alerts,
# 2026-10-22; two critical JPEG decoder overflows), the same decoder
# overflow / DoS classes as 12.3.0's: cover_hash.py feeds it remote cover
# URLs, so they are in scope. 13.0's breaking changes (Python 3.10 dropped,
# I-mode PNG saves, fromarray mode, a few removed attributes) touch none of
# the API used here (Image.open + UnidentifiedImageError + imagehash.phash):
# the cover-hash tests pass unchanged, and 3 of 120 synthetic covers' pHashes
# move by 2 bits, inside the 0-6 match band.
Pillow==13.0.0
ImageHash==4.3.2

# v2.11.0 — TLS-fingerprint impersonation for sources fronted by
# Akamai Bot Manager (Amazon), AWS WAF (Goodreads) and Cloudflare (Kobo). curl_cffi drives libcurl-impersonate
# which exactly replicates Chrome's JA3 fingerprint at the protocol
# layer, which Python's standard TLS (httpx, requests) never matches.
# AmazonSource uses `AsyncSession(impersonate="chrome120")`; Phase 5.5
# Goodreads source-scan path will use the same lib + `impersonate`.
curl_cffi>=0.7.0

# Fernet symmetric encryption for the secret store.
# Patch cascade: 46.0.5 fixes CVE-2026-26007 (SECT-curve subgroup
# validation, HIGH); 46.0.6 rolls in CVE-2026-34073 (DNS name
# constraint enforcement) and a patched OpenSSL wheel (older
# CVE-2024-12797); 46.0.7 closes CVE-2026-39892 (buffer overflow
# when non-contiguous Python buffers are passed to Hash-family APIs).
# 49.0.0 clears GHSA-537c-gmf6-5ccf (HIGH): the 46.x wheels statically
# bundle a vulnerable OpenSSL; 49.0.0 ships a patched one. We only use
# Fernet (symmetric AES/HMAC) — no untrusted TLS/X.509 through this lib,
# so practical exposure was low, but a runtime dep is worth keeping clean.
# Fernet wire format + API are unchanged across 46→50 — no code impact,
# and existing encrypted secrets decrypt as-is.
cryptography==50.0.0
